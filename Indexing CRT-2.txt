TASK 7: OPTIMIZING INDEX ORDER FOR PERFORMANCE
=====================================================================

studentdb> // Rule (ESR): Equality fields first, then Sort fields, then Range fields.
studentdb> // Query: course = "BCA", sort by age, range on marks > 60

studentdb> // ---- Bad order: Range first ----
studentdb> db.students.createIndex({ marks: 1, age: 1, course: 1 }, { name: "bad_order" })
bad_order

studentdb> db.students.find({ course: "BCA", marks: { $gt: 60 } }).sort({ age: 1 }).hint("bad_order").explain("executionStats").executionStats
{
  nReturned: 7587,
  executionTimeMillis: 38,
  totalKeysExamined: 18290,
  totalDocsExamined: 12460
}

studentdb> // ---- Good order: Equality, Sort, Range (ESR) ----
studentdb> db.students.createIndex({ course: 1, age: 1, marks: 1 }, { name: "good_order_ESR" })
good_order_ESR

studentdb> db.students.find({ course: "BCA", marks: { $gt: 60 } }).sort({ age: 1 }).hint("good_order_ESR").explain("executionStats").executionStats
{
  nReturned: 7587,
  executionTimeMillis: 14,
  totalKeysExamined: 9210,
  totalDocsExamined: 7587
}

studentdb> // Check there is NO in-memory SORT stage in the good plan
studentdb> db.students.find({ course: "BCA", marks: { $gt: 60 } }).sort({ age: 1 }).hint("good_order_ESR").explain().queryPlanner.winningPlan
{
  stage: 'FETCH',
  inputStage: {
    stage: 'IXSCAN',
    keyPattern: { course: 1, age: 1, marks: 1 },
    indexName: 'good_order_ESR',
    direction: 'forward'
  }
}
studentdb> // No SORT stage --> sorting is done by the index itself

studentdb> // ---- Prefix rule demo ----
studentdb> // Index {course:1, age:1, marks:1} supports:
studentdb> //   { course }
studentdb> //   { course, age }
studentdb> //   { course, age, marks }
studentdb> // It does NOT efficiently support { age } or { marks } alone.
studentdb> db.students.find({ marks: 70 }).explain().queryPlanner.winningPlan.inputStage.indexName
marks_-1

studentdb> // ---- Sort direction check ----
studentdb> db.students.find().sort({ course: 1, marks: -1 }).explain().queryPlanner.winningPlan.inputStage.indexName
course_1_marks_-1
studentdb> db.students.find().sort({ course: 1, marks: 1 }).explain().queryPlanner.winningPlan.stage

studentdb> // Mixed directions must match the index (or its exact reverse), otherwise a SORT stage appears.

studentdb> // ---- Clean up the bad index (hide first, then drop) ----
